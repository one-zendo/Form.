function localDay(d){return d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0")}function weekStart(value){const d=new Date(value);d.setHours(0,0,0,0);d.setDate(d.getDate()-(d.getDay()+6)%7);return d}function shiftDay(d,n){const next=new Date(d);next.setDate(next.getDate()+n);return next}function validWeekKey(key){if(typeof key!=="string"||!/^\d{4}-\d{2}-\d{2}$/.test(key))return false;const d=new Date(key+"T12:00:00");return Number.isFinite(d.getTime())&&localDay(d)===key&&d.getDay()===1}function validSchedules(v){return Array.isArray(v)&&v.every((x,i)=>x&&validWeekKey(x.from)&&(!i||v[i-1].from<x.from)&&Array.isArray(x.rest)&&x.rest.length<7&&new Set(x.rest).size===x.rest.length&&x.rest.every(n=>Number.isInteger(n)&&n>=0&&n<7))}
function calculateStreak(now,records,plans){const current=weekStart(now),key=localDay(current),daysByWeek=new Map(),scheduleAt=key=>plans.filter(s=>s.from<=key).at(-1);for(const h of records){if(h.finished>now.getTime()||!h.exercises.some(e=>e.sets.some(t=>t.done)))continue;const date=new Date(h.finished),week=localDay(weekStart(date));if(!daysByWeek.has(week))daysByWeek.set(week,new Set());daysByWeek.get(week).add(localDay(date))}let run=0,best=0;const first=plans[0]?.from;if(first){let previous=null;for(const [week,days] of [...daysByWeek].sort((a,b)=>a[0].localeCompare(b[0]))){if(week<first||week>key)continue;const plan=scheduleAt(week);if(!plan||days.size<7-plan.rest.length){if(week!==key){run=0;previous=null}continue}const date=new Date(week+"T12:00:00");run=previous===localDay(shiftDay(date,-7))?run+1:1;best=Math.max(best,run);previous=week}if(previous!==key&&previous!==localDay(shiftDay(current,-7)))run=0}return {current,key,plan:scheduleAt(key),days:daysByWeek.get(key)||new Set(),run,best}}
