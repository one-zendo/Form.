function recordStats(record){const sets=record.exercises.flatMap(e=>e.sets).filter(t=>t.done);return {sets:sets.length,reps:sets.reduce((n,t)=>n+t.reps,0),volume:sets.reduce((n,t)=>n+t.weight*t.reps,0),minutes:Math.max(0,Math.round((record.finished-record.started)/60000)),exercises:record.exercises.filter(e=>e.sets.some(t=>t.done)).length}}
function progressRecords(records,period,now){const days=Number(period),cutoff=period==="all"||!Number.isFinite(days)||days<=0?-Infinity:now-days*86400000;return records.filter(h=>h.finished>=cutoff&&h.finished<=now&&h.exercises.some(e=>e.sets.some(t=>t.done))).sort((a,b)=>a.finished-b.finished)}function exerciseSeries(records){const map=new Map();for(const h of [...records].sort((a,b)=>a.finished-b.finished)){const tops=new Map();for(const e of h.exercises){const sets=e.sets.filter(t=>t.done);if(!sets.length)continue;tops.set(e.name,Math.max(tops.get(e.name)||0,...sets.map(t=>t.weight)))}for(const [name,weight] of tops){const arr=map.get(name)||[];arr.push({date:h.finished,weight});map.set(name,arr)}}return map}
function progressChartPoints(points,min,max){const first=points[0].date,last=points.at(-1).date,span=max-min||1;return points.map((p,i)=>(8+304*(last===first?i/Math.max(1,points.length-1):(p.date-first)/(last-first)))+","+(max===min?50:92-(p.weight-min)/span*84)).join(" ")}
function savedWeightRecords(record,prior){const previous=new Map(),current=new Map();for(const h of prior)if(h.finished<=record.finished&&h.id!==record.id)for(const e of h.exercises)for(const t of e.sets)if(t.done)previous.set(e.name,Math.max(previous.get(e.name)??-Infinity,t.weight));for(const e of record.exercises)for(const t of e.sets)if(t.done)current.set(e.name,Math.max(current.get(e.name)??-Infinity,t.weight));return [...current].filter(([name,weight])=>previous.has(name)&&weight>previous.get(name)).map(([name,weight])=>({name,weight,from:previous.get(name)}))}
function buildPreviousSetIndex(records){const result=new Map();for(const h of [...records].sort((a,b)=>b.finished-a.finished))for(const e of h.exercises){let sets=result.get(e.name);if(!sets){sets=new Map();result.set(e.name,sets)}for(const [i,t] of e.sets.entries())if(t.done&&!sets.has(i))sets.set(i,t)}return result}
