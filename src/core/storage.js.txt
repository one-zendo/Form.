function validCleanSlate(p){return !!p&&uniqueRecordIds(p.records)&&p.records.length===48&&p.records.every(r=>validRoutine(r)&&r.id==="clean-slate-"+r.finished&&r.description==="Synthetic development workout"&&validDateStamp(r.started)&&validDateStamp(r.finished)&&r.finished>r.started&&r.exercises.length>0&&r.exercises.every(e=>e.sets.length>0&&e.sets.every(s=>s.done===true)))&&validSchedules(p.schedules)&&p.schedules.length===1}
function completeCleanSlate(storage){
