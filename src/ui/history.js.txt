const historyDateOptions={weekday:"long",month:"short",day:"numeric"};
const historyDates={
 current:new Intl.DateTimeFormat(undefined,historyDateOptions),
 withYear:new Intl.DateTimeFormat(undefined,{...historyDateOptions,year:"numeric"}),
 long:new Intl.DateTimeFormat(undefined,{...historyDateOptions,year:"numeric",month:"long"}),
 numeric:new Intl.DateTimeFormat(),
 detail:new Intl.DateTimeFormat(undefined,{dateStyle:"medium",timeStyle:"short"})
};
const historySearchIndex=new WeakMap();
function historySearchText(record){if(historySearchIndex.has(record))return historySearchIndex.get(record);const date=new Date(record.finished);const parts=[record.name,...record.exercises.map(e=>e.name),historyDates.long.format(date),historyDates.withYear.format(date),historyDates.numeric.format(date),[date.getFullYear(),String(date.getMonth()+1).padStart(2,"0"),String(date.getDate()).padStart(2,"0")].join("-")];const text=parts.join(" ").toLocaleLowerCase();historySearchIndex.set(record,text);return text}

function filteredHistory(){const terms=workoutQueries.history.trim().toLocaleLowerCase().split(/\s+/).filter(Boolean);return [...history].filter(record=>{if(!terms.length)return true;const haystack=historySearchText(record);return terms.every(term=>haystack.includes(term))}).sort((a,b)=>b.finished-a.finished)}

function renderHistory(){const root=$("workout-history-list");root.replaceChildren();const currentYear=new Date().getFullYear(),matches=filteredHistory();setText(historyCount,matches.length+" workout"+(matches.length===1?"":"s"));for(const record of matches){const card=action("",()=>{selectedHistoryId=record.id;if(wide.matches){syncHistorySelection();renderHistoryDetail(record,true)}else showSummary(record)});card.className="card history-entry workout-history-card";card.dataset.historyId=record.id;const finished=new Date(record.finished);const date=el("span",(finished.getFullYear()===currentYear?historyDates.current:historyDates.withYear).format(finished),"caption history-date");const stats=recordStats(record);const copy=el("div","","history-copy");copy.append(el("strong",record.name),el("span",historyDuration(record,stats)+" · "+stats.sets+" sets","caption"));card.append(copy,date);root.append(card)}if(!matches.length){const empty=el("article","","card");empty.append(el("h3",sessionBlocked?"Workout records need recovery.":history.length?"No matching workouts.":"No recorded workouts yet."),el("p",sessionBlocked?"Reload FORM to recover your saved records.":history.length?"Try another search.":"Complete your first workout to see it here.","muted"));root.append(empty)}if(workoutsSection==="history")layoutHistoryDetail()}

function syncHistorySelection(){for(const card of $("workout-history-list").querySelectorAll(".workout-history-card")){const selected=wide.matches&&card.dataset.historyId===selectedHistoryId;card.classList.toggle("selected",selected);if(selected)card.setAttribute("aria-current","true");else card.removeAttribute("aria-current")}}

function populateHistoryStats(root,record){const stats=recordStats(record);root.replaceChildren();root.classList.add("history-unified-stats");for(const [name,value] of [["Duration",historyDuration(record)],["Total volume",numberLabel(stats.volume)+" kg"],["Exercises",stats.exercises],["Completed sets",stats.sets]])addStat(root,name,value)}

function historyTrend(record,exercise){
 const panel=el("section","","history-trend"),points=historyExercisePoints(record,exercise.name,history);if(points.length<2){panel.hidden=true;return panel}
 const controls=el("div","","history-metric-control"),output=el("div","","history-chart-output");controls.setAttribute("role","group");controls.setAttribute("aria-label","Exercise trend metric");const metrics=[["weight","Weight"],["volume","Volume"],["estimated","Est. 1RM"]];let metric="weight";const buttons=metrics.map(([key,label])=>{const button=action(label,()=>{metric=key;draw()});controls.append(button);return button});panel.append(controls,output);
 function draw(){buttons.forEach((b,i)=>b.setAttribute("aria-pressed",String(metrics[i][0]===metric)));output.replaceChildren();const label=metrics.find(m=>m[0]===metric)[1],values=points.filter(p=>Number.isFinite(p[metric])),latest=points.at(-1);if(metric==="estimated"){const explanation=el("details","","history-estimate-info");explanation.append(el("summary","About Est. 1RM"),el("p","Estimated maximum single-rep lift, not a tested maximum. Epley: weight × (1 + reps / 30), for 2–10 reps; single reps use actual weight.","caption"));output.append(explanation)}if(!Number.isFinite(latest[metric])||values.length<2){output.append(el("p","Not enough eligible sessions for this metric.","muted"));return}
 const first=values[0],delta=latest[metric]-first[metric],headline=el("div","","row history-metric-value");headline.append(el("strong",numberLabel(latest[metric])+" kg"));if(first[metric]>0)headline.append(el("span",(delta>0?"+":"")+numberLabel(delta/first[metric]*100)+"% · displayed period",delta>0?"history-improved":delta<0?"history-declined":"muted"));output.append(headline);
 const ns="http://www.w3.org/2000/svg",svg=document.createElementNS(ns,"svg");svg.setAttribute("viewBox","0 0 320 100");svg.setAttribute("class","history-chart");svg.setAttribute("role","img");svg.setAttribute("aria-label",label+" in kilograms: "+values.map(p=>new Date(p.date).toLocaleDateString()+": "+numberLabel(p[metric])).join("; "));
 const shape=(tag,attrs)=>{const n=document.createElementNS(ns,tag);for(const [k,v] of Object.entries(attrs))n.setAttribute(k,String(v));return n};for(const y of [8,36,64,92])svg.append(shape("line",{x1:8,x2:312,y1:y,y2:y,stroke:"var(--line)","stroke-dasharray":"3 4"}));
 const min=Math.min(...values.map(p=>p[metric])),max=Math.max(...values.map(p=>p[metric])),coords=progressChartPoints(values.map(p=>({date:p.date,weight:p[metric]})),min,max);svg.append(shape("polygon",{points:"8,96 "+coords+" 312,96",fill:"var(--workout-accent)",opacity:".08"}),shape("polyline",{points:coords,fill:"none",stroke:"var(--workout-accent)","stroke-width":2,"stroke-linejoin":"round","stroke-linecap":"round"}));const xy=coords.split(" ").map(p=>p.split(","));xy.forEach(([x,y],i)=>svg.append(shape("circle",{cx:x,cy:y,r:i===xy.length-1?4:2.5,fill:"var(--workout-accent)"})));output.append(svg);
 const dates=el("div","","row caption");dates.append(el("span",new Date(first.date).toLocaleDateString(undefined,{month:"short",day:"numeric"})),el("span",new Date(latest.date).toLocaleDateString(undefined,{month:"short",day:"numeric"})));output.append(dates);
 const picker=el("select","","history-chart-inspect");picker.setAttribute("aria-label","Inspect recorded "+label+" session");for(const p of values){const option=el("option",new Date(p.date).toLocaleDateString()+" · "+numberLabel(p[metric])+" kg");option.value=p.id;picker.append(option)}picker.value=latest.id;const inspected=el("p","","caption");inspected.setAttribute("aria-live","polite");picker.onchange=()=>{const p=values.find(p=>p.id===picker.value);setText(inspected,p?label+": "+numberLabel(p[metric])+" kg":"")};picker.onchange();output.append(picker,inspected);if(motionAllowed())formAnimate(output,[{opacity:.6},{opacity:1}],formMotion.fast)}
 draw();return panel;
}

function renderHistoryExercises(root,record){root.replaceChildren();const prior=history.filter(h=>h.finished<record.finished&&h.id!==record.id),records=savedWeightRecords(record,prior);
 for(const exercise of record.exercises){const done=exercise.sets.filter(t=>t.done);if(!done.length)continue;const card=el("details","","card detail-exercise history-exercise-card"),summary=el("summary"),label=el("span","","history-exercise-copy"),volume=done.reduce((n,t)=>n+t.weight*t.reps,0);label.append(el("strong",exercise.name),el("span",done.length+(done.length===1?" set":" sets")+" · "+numberLabel(volume)+" kg","muted"));const pr=records.find(p=>p.name===exercise.name);if(pr)label.append(el("span","Personal record · "+numberLabel(pr.weight)+" kg","caption history-improved"));const chevron=el("span","","detail-chevron icon");const arrow=document.querySelector("#personal-open > .icon svg");if(arrow)chevron.append(arrow.cloneNode(true));summary.append(label,chevron);card.append(summary);const content=el("div","","history-exercise-content"),table=el("div","","history-set-table"),columns=el("div","","history-set-row caption");for(const label of ["Set","Previous","kg","Reps"])columns.append(el("span",label));table.append(columns);const previous=historyPreviousSets(record,exercise.name,history);exercise.sets.forEach((set,i)=>{if(!set.done)return;const row=el("div","","history-set-row"),p=previous.get(i),improved=!!p&&set.weight>=p.weight&&set.reps>=p.reps&&(set.weight>p.weight||set.reps>p.reps);row.append(el("span",String(i+1)),el("span",p?numberLabel(p.weight)+" × "+p.reps:"—","muted"),el("span",numberLabel(set.weight),improved?"history-improved":""),el("span",String(set.reps),improved?"history-improved":""));table.append(row)});const points=historyExercisePoints(record,exercise.name,history);if(points.length>1){const diff=volume-points.at(-2).volume;table.append(el("p",(diff>0?"+":"")+numberLabel(diff)+" kg volume vs previous",diff>0?"caption history-improved":"caption muted"))}content.append(table,historyTrend(record,exercise));card.append(content);animateRoutineCard(card);root.append(card)}
 if(!root.children.length)root.append(el("p","No completed sets.","muted"));
}

function renderHistoryDetail(record,animate=false){if(record&&record===renderedHistoryRecord)return;renderedHistoryRecord=record;historyHeading.textContent=record?.name||"Workout details";historyDate.hidden=!record;historyScroll.replaceChildren();historyScroll.scrollTop=0;if(!record){const empty=el("article","","card");empty.append(el("h3",sessionBlocked?"Workout records need recovery.":history.length?"No matching workouts":"No recorded workouts yet"),el("p",sessionBlocked?"Reload FORM to recover your saved records.":history.length?"Try a different search.":"Your completed workouts and exercise details will appear here.","muted"));historyScroll.append(empty);return}historyDate.textContent=historyDates.detail.format(new Date(record.finished));const statGrid=el("div","","history-detail-stats");populateHistoryStats(statGrid,record);historyScroll.append(statGrid);const heading=el("div","","row detail-heading");heading.append(el("h3","Exercises"));historyScroll.append(heading);const list=el("div","","history-detail-exercises");renderHistoryExercises(list,record);historyScroll.append(list);if(animate)formAnimate(historyBody,[{opacity:.55},{opacity:1}],formMotion.fast)}

function layoutHistoryDetail(){const visible=workoutsSection==="history"&&wide.matches;historyDetail.hidden=!visible;if(visible){const matches=filteredHistory();const record=matches.find(r=>r.id===selectedHistoryId)||matches[0]||null;selectedHistoryId=record?.id||null;renderHistoryDetail(record)}syncHistorySelection()}
